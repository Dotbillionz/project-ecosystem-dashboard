export async function runTrade() {

  const symbol = 'BTC/USDT';

  // 💰 GET BALANCE
  const balance = await exchange.fetchBalance();
  const usdt = balance.total.USDT;

  // 📊 GET PRICE
  const ticker = await exchange.fetchTicker(symbol);
  const price = ticker.last;

  // 📉 SIMPLE TREND FILTER (DON’T TRADE RANDOMLY)
  const ohlcv = await exchange.fetchOHLCV(symbol, '1m', undefined, 20);
  const closes = ohlcv.map(c => c[4]);

  const avg = closes.reduce((a,b)=>a+b,0)/closes.length;

  if (price < avg) {
    console.log("Downtrend — skipping trade");
    return;
  }

  // 📦 POSITION SIZE (RISK BASED)
  const riskAmount = usdt * RISK_PER_TRADE;
  const stopLossDistance = price * STOP_LOSS_PERCENT;

  const amount = riskAmount / stopLossDistance;

  console.log("Buying amount:", amount);

  // 🟢 BUY
  const order = await exchange.createMarketBuyOrder(symbol, amount);

  const entryPrice = price;

  // 🔻 STOP LOSS
  const stopPrice = entryPrice * (1 - STOP_LOSS_PERCENT);

  // 🔺 TAKE PROFIT
  const takeProfit = entryPrice * (1 + TAKE_PROFIT_PERCENT);

  console.log("SL:", stopPrice);

const app = express();

let users = {};

// 🔐 REGISTER
app.post("/auth/register", async (req, res) => {
  const { email, password } = req.body;

  if (users[email]) return res.status(400).json({ error: "User exists" });

  const hashed = await bcrypt.hash(password, 10);

  users[email] = {
    password: hashed,
    balance: 1000,
    pnl: 0,
    copy: true
  };

  res.json({ message: "Registered" });
});

// 🔐 LOGIN
app.post("/auth/login", async (req, res) => {
  const { email, password } = req.body;

  const user = users[email];
  if (!user) return res.status(404).json({ error: "User not found" });

  const match = await bcrypt.compare(password, user.password);
  if (!match) return res.status(401).json({ error: "Wrong password" });

  const token = jwt.sign({ email }, SECRET);

  res.json({ token });
});

// 🔐 AUTH MIDDLEWARE
function auth(req, res, next) {
  const token = req.headers.authorization;
  if (!token) return res.sendStatus(403);

  try {
    const decoded = jwt.verify(token, SECRET);
    req.user = decoded.email;
    next();
  } catch {
    res.sendStatus(403);
  }
}

// 💳 GET DASHBOARD DATA
app.get("/dashboard", auth, (req, res) => {
  const user = users[req.user];
  res.json({
    balance: user.balance,
    pnl: user.pnl,
    copy: user.copy
  });
});

// 🔁 TOGGLE COPY
app.post("/user/toggle-copy", auth, (req, res) => {
  users[req.user].copy = !users[req.user].copy;
  res.json({ copy: users[req.user].copy });
