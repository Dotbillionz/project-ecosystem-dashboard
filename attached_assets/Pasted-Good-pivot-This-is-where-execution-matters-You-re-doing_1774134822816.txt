# 🧠 STEP 3 — CONNECT TRADING ENGINE

### `tradebot.js`

```javascript
import ccxt from 'ccxt';

const exchange = new ccxt.binance({
  apiKey: 'YOUR_API_KEY',
  secret: 'YOUR_SECRET',
  enableRateLimit: true
});

export async function runTrade() {

  const symbol = 'BTC/USDT';

  const ticker = await exchange.fetchTicker(symbol);

  console.log("Price:", ticker.last);

  // Example strategy (basic)
  if (ticker.last < 60000) {
    await exchange.createMarketBuyOrder(symbol, 0.001);
    console.log("BUY executed");
  } else {
    await exchange.createMarketSellOrder(symbol, 0.001);
    console.log("SELL executed");
  }

}
```

# 🔌 STEP 4 — CONNECT TO YOUR API
