<script>
function openView(id){
  document.getElementById("home").style.display="none";
  document.querySelectorAll(".view").forEach(v=>v.style.display="none");
  document.getElementById(id).style.display="block";
}

function goHome(){
  document.querySelectorAll(".view").forEach(v=>v.style.display="none");
  document.getElementById("home").style.display="grid";
}

function goWallet(){
  window.location.href="https://spotxwallet.com";
}

// 🔌 BACKEND CONNECTIONS
async function liveTrade(){
  await fetch("/trade/live",{method:"POST"});
  alert("Trade executed");
}

async function deposit(){
  window.location.href="https://pay.sumup.com/b2c/YOUR_LINK";
}

  const res = await fetch("/wallet/user1");
  const data = await res.json();
  alert("Balance: €"+data.balance);
}

  await fetch("/user/toggle-copy",{method:"POST"});
  alert("Copy trading toggled");
}
